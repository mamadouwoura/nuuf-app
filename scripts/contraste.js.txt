// Audit de contraste (WCAG) à coller dans la console du navigateur : renvoie les textes sous 4.5:1 (3:1 pour les grands textes).
(async () => {
  // Attend que la page soit complètement affichée (le serveur de développement compile à la demande).
  let prec = -1, stable = 0;
  for (let i = 0; i < 40 && stable < 3; i++) {
    await new Promise((r) => setTimeout(r, 250));
    const l = document.body.innerText.length;
    stable = l === prec && l > 120 ? stable + 1 : 0;
    prec = l;
  }
  await new Promise((r) => setTimeout(r, 700));
  const parse = (c) => {
    const m = c.match(/rgba?\(([^)]+)\)/);
    if (!m) return null;
    const p = m[1].split(/[ ,\/]+/).filter(Boolean).map(Number);
    return { r: p[0], g: p[1], b: p[2], a: p.length > 3 ? p[3] : 1 };
  };
  const lum = ({ r, g, b }) => {
    const f = (v) => { v /= 255; return v <= 0.03928 ? v / 12.92 : Math.pow((v + 0.055) / 1.055, 2.4); };
    return 0.2126 * f(r) + 0.7152 * f(g) + 0.0722 * f(b);
  };
  const melange = (haut, bas) => ({ r: haut.r * haut.a + bas.r * (1 - haut.a), g: haut.g * haut.a + bas.g * (1 - haut.a), b: haut.b * haut.a + bas.b * (1 - haut.a), a: 1 });
  const fond = (el) => {
    const pile = [];
    for (let e = el; e; e = e.parentElement) {
      const bg = parse(getComputedStyle(e).backgroundColor);
      if (bg && bg.a > 0) { pile.push(bg); if (bg.a >= 1) break; }
    }
    let c = parse(getComputedStyle(document.body).backgroundColor);
    if (!c || c.a < 1) c = { r: 255, g: 255, b: 255, a: 1 };
    if (pile.length && pile[pile.length - 1].a >= 1) c = pile.pop();
    while (pile.length) c = melange(pile.pop(), c);
    return c;
  };
  const res = [];
  const vus = new Set();
  const w = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT);
  while (w.nextNode()) {
    const n = w.currentNode;
    const txt = n.textContent.trim();
    if (!txt || !/[A-Za-zÀ-ÿ0-9]/.test(txt)) continue;
    const el = n.parentElement;
    if (!el || vus.has(el)) continue;
    const cs = getComputedStyle(el);
    if (cs.visibility === "hidden" || cs.display === "none" || parseFloat(cs.opacity) === 0) continue;
    const r = el.getBoundingClientRect();
    if (r.width === 0 || r.height === 0) continue;
    if (el.closest(".material-symbols-outlined")) continue;
    vus.add(el);
    let col = parse(cs.color);
    if (!col) continue;
    const bg = fond(el);
    if (col.a < 1) col = melange(col, bg);
    // opacité héritée
    let op = 1;
    for (let e = el; e; e = e.parentElement) op *= parseFloat(getComputedStyle(e).opacity);
    if (op < 1) col = melange({ ...col, a: op }, bg);
    const L1 = lum(col), L2 = lum(bg);
    const ratio = (Math.max(L1, L2) + 0.05) / (Math.min(L1, L2) + 0.05);
    const taille = parseFloat(cs.fontSize), gras = parseInt(cs.fontWeight) >= 700;
    const grand = taille >= 24 || (taille >= 18.66 && gras);
    const seuil = grand ? 3 : 4.5;
    if (ratio < seuil) res.push({ txt: txt.slice(0, 40), ratio: Math.round(ratio * 100) / 100, fg: `rgb(${Math.round(col.r)},${Math.round(col.g)},${Math.round(col.b)})`, bg: `rgb(${Math.round(bg.r)},${Math.round(bg.g)},${Math.round(bg.b)})`, cls: (el.className || "").toString().slice(0, 60) });
  }
  // Textes qui ressemblent encore à du français (utile en mode anglais)
  const MOTS = /(^|[^a-z\u00e0-\u00ff])(le|la|les|des|du|de|et|sur|pour|avec|votre|vos|une|dans|est|ce|cette|qui|pas|aux|par|ou|mois|loyers?|biens?|locataires?|quittances?|paiements?|aucun|aucune|tous|toutes|retard|attente|jours?|bailleurs?|forfaits?|janvier|f\u00e9vrier|mars|avril|mai|juin|juillet|ao\u00fbt|septembre|octobre|novembre|d\u00e9cembre|sans|tout|plus|moins|chaque|gratuit|essai)([^a-z\u00e0-\u00ff]|$)/i;
  const fr = [];
  const w2 = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT);
  while (w2.nextNode()) {
    const t = w2.currentNode.textContent.trim(), el = w2.currentNode.parentElement;
    if (!t || t.length < 3 || !el || el.closest("script,style,noscript,.material-symbols-outlined")) continue;
    if (/[éèêëàâùûôîïç]/i.test(t) || MOTS.test(t)) fr.push(t.replace(/\s+/g, " ").slice(0, 60));
  }
  return { p: location.pathname + location.search, n: res.length, contraste: res.slice(0, 6).map((o) => o.ratio + " " + o.txt + " [" + o.fg + " on " + o.bg + "]"), fr: [...new Set(fr)].slice(0, 12) };
})();
